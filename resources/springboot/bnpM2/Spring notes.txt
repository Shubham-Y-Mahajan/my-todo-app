package com.bnp.service;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.context.annotation.Lazy;
import org.springframework.stereotype.Service;

import com.bnp.model.Product;
import com.bnp.repository.ProductRepository;

@Service
public class ProductService {
	
	@Autowired
	private ProductRepository productRepository;
	
	@Autowired
	@Qualifier("festivalDiscount")
	@Lazy
	private DiscountService discountService;
	
	public Product fetchProduct() {
		Product prod = productRepository.getProduct();
		discountService.applyDiscount();
		return prod;
	}
	
	

}

pipeline {
    agent any 
    tools{
	jdk 'Java17'
        maven 'Maven'
    }
    stages {
        stage('Checkout Code from Github') {
            steps {
                echo 'Pull the application from github' 
                git branch: 'main', credentialsId: 'gitcred', url: 'https://github.com/chntraining/bnptest.git'
            }
        }

        stage('Building the Project') {
            steps {
                echo 'Build the application ' 
                bat 'mvn clean package'
            }
        }

	stage('Build Docker Image') {
            steps {
                echo 'Build the Docker Image for the application ' 
                bat 'docker build -t pms-app:latest .'
            }
        }
        stage('Start the minikube') {
            steps {
		echo 'Deleting Minikube' 
		bat 'minikube delete'
                echo 'Starting Minikube' 
                bat 'minikube start'
            }
        }
        stage('Load the image to Minikube') {
            steps {
                echo 'Loading image to minikube ' 
                bat 'minikube image load pms-app:latest'
            }
        }

	stage('Deploy to k8') {
            steps {
                echo 'Deploying to K8s' 
                bat 'kubectl apply -f k8s/'

		echo 'Checking Pods' 
                bat 'kubectl get pods'

		echo 'Checking Services' 
                bat 'kubectl get svc'
            }
        }

	 stage('Wait for the pods to be ready') {
            steps {
                echo 'Waiting for the pods to be ready' 
                bat 'kubectl wait --for=condition=Ready pods --all  --timeout=120s'
            }
         }

	stage('Access the application') {
            steps {
                echo 'Accessing Service URL' 
                bat 'minikube service pms-service --url'
            }
         }

    }
}
